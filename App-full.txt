import React,{useEffect,useMemo,useState} from 'react';
import {api,token} from './api';
import {Activity,LayoutDashboard,ScanSearch,History,Lightbulb,MessageCircle,Settings,Sun,Moon,LogOut,Globe,Menu,PlayCircle,FileDown,Trash2,ArrowUpRight,Smile,Meh,Frown,Send,LoaderCircle,ExternalLink} from 'lucide-react';
import {ResponsiveContainer,PieChart,Pie,Cell,Tooltip,BarChart,Bar,XAxis,YAxis,CartesianGrid,LineChart,Line} from 'recharts';

const demoHelp="Demo mode uses synthetic example comments, not data retrieved from your submitted social media post.";
function App(){
 const [auth,setAuth]=useState(!!token()),[user,setUser]=useState(null),[page,setPage]=useState('Dashboard');
 const [dark,setDark]=useState(localStorage.getItem('sp_theme')==='dark'),[lang,setLang]=useState(localStorage.getItem('sp_lang')||'en');
 const [email,setEmail]=useState(''),[password,setPassword]=useState(''),[name,setName]=useState('');
 const [url,setUrl]=useState(''),[demo,setDemo]=useState(true),[loading,setLoading]=useState(false),[error,setError]=useState(''),[result,setResult]=useState(null);
 const [history,setHistory]=useState([]),[stats,setStats]=useState(null),[chat,setChat]=useState([]),[chatText,setChatText]=useState('');
 const [onboarding,setOnboarding]=useState(false),[step,setStep]=useState(0),[selected,setSelected]=useState(null);
 const t=(en,hi)=>lang==='hi'?hi:en;
 useEffect(()=>{document.documentElement.classList.toggle('dark',dark);localStorage.setItem('sp_theme',dark?'dark':'light')},[dark]);
 useEffect(()=>{localStorage.setItem('sp_lang',lang)},[lang]);
 useEffect(()=>{if(auth){loadAll()}},[auth]);
 async function loadAll(){
  try{const [p,h,s]=await Promise.all([api('/auth/profile'),api('/analysis/history'),api('/analytics/dashboard')]);setUser(p);setHistory(h);setStats(s);if(!p.onboarding_completed){setOnboarding(true);setStep(0)}}catch(e){localStorage.removeItem('sp_token');setAuth(false)}
 }
 async function authenticate(mode){
  setError('');try{const d=await api('/auth/'+mode,{method:'POST',body:JSON.stringify({name,email,password})});localStorage.setItem('sp_token',d.token);setUser(d.user);setAuth(true)}catch(e){setError(e.message)}
 }
 async function analyze(e){e?.preventDefault();setError('');setResult(null);setLoading(true);
  try{const r=await api('/analysis/analyze',{method:'POST',body:JSON.stringify({url,demo})});setResult(r);setSelected(r);setPage('Analysis');await loadAll()}catch(e){setError(e.message)}finally{setLoading(false)}
 }
 async function openReport(id){try{const r=await api('/analysis/'+id);setResult(r);setSelected(r);setPage('Analysis')}catch(e){setError(e.message)}}
 async function remove(id){try{await api('/analysis/'+id,{method:'DELETE'});await loadAll();if(result?.id===id)setResult(null)}catch(e){setError(e.message)}}
 async function askBot(e){e?.preventDefault();if(!chatText.trim())return;const q=chatText;setChat(c=>[...c,{role:'user',text:q}]);setChatText('');try{const r=await api('/chatbot/message',{method:'POST',body:JSON.stringify({message:q,analysis_id:result?.id})});setChat(c=>[...c,{role:'assistant',text:r.response}])}catch(err){setChat(c=>[...c,{role:'assistant',text:err.message}])}}
 async function finishTour(){setOnboarding(false);try{await api('/settings',{method:'PUT',body:JSON.stringify({onboarding_completed:true})})}catch{}}
 async function downloadPdf(id){try{const res=await api('/reports/'+id+'/pdf',{raw:true});const blob=await res.blob();const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download=`socialpulse-report-${id}.pdf`;a.click();URL.revokeObjectURL(a.href)}catch(e){setError(e.message)}}
 const pie=result?[{name:'Positive',value:result.positive_count,color:'#16a34a'},{name:'Negative',value:result.negative_count,color:'#ef4444'},{name:'Neutral',value:result.neutral_count,color:'#64748b'}]:[];
 const nav=[['Dashboard',LayoutDashboard],['New Analysis',ScanSearch],['Analysis History',History],['AI Recommendations',Lightbulb],['AI Assistant',MessageCircle],['Settings',Settings]];
 if(!auth)return <div className="auth-shell"><div className="auth-card"><div className="brand"><span className="brand-icon"><Activity/></span><div><b>SocialPulse AI</b><small>Social intelligence, made clear.</small></div></div><div className="auth-copy"><span className="eyebrow">AI-POWERED SOCIAL LISTENING</span><h1>Understand the conversation.</h1><p>Turn public feedback into clear sentiment insights and practical decisions.</p></div><div className="auth-form"><h2>Welcome back</h2><p className="muted">Sign in or create your account to continue.</p><label>Full name (for registration)</label><input placeholder="Your name" value={name} onChange={e=>setName(e.target.value)}/><label>Email address</label><input type="email" placeholder="you@example.com" value={email} onChange={e=>setEmail(e.target.value)}/><label>Password</label><input type="password" placeholder="At least 8 characters" value={password} onChange={e=>setPassword(e.target.value)}/>{error&&<div className="error">{error}</div>}<button className="primary full" onClick={()=>authenticate('login')}>Sign in</button><button className="secondary full" onClick={()=>authenticate('register')}>Create account</button><small className="muted">Registration requires a name, email and password. Password reset email delivery needs to be configured separately.</small></div></div></div>;
 return <div className="app-shell">
  <aside className="sidebar"><div className="brand"><span className="brand-icon"><Activity/></span><div><b>SocialPulse AI</b><small>Social intelligence</small></div></div><div className="nav-label">WORKSPACE</div>{nav.map(([n,I])=><button key={n} onClick={()=>setPage(n)} className={'nav-item '+(page===n?'active':'')}><I size={18}/>{n}</button>)}<div className="sidebar-bottom"><div className="user-mini"><div className="avatar">{(user?.name||'U')[0].toUpperCase()}</div><div><b>{user?.name||'User'}</b><small>{user?.email}</small></div></div><button className="nav-item" onClick={()=>{localStorage.removeItem('sp_token');setAuth(false)}}><LogOut size={17}/>Log out</button></div></aside>
  <main className="main"><header className="topbar"><div><span className="eyebrow">SOCIAL MEDIA INTELLIGENCE</span><h2>{page}</h2></div><div className="top-actions"><select value={lang} onChange={e=>setLang(e.target.value)}><option value="en">English</option><option value="hi">हिन्दी</option></select><button className="icon-btn" title="Toggle theme" onClick={()=>setDark(!dark)}>{dark?<Sun size={18}/>:<Moon size={18}/>}</button></div></header>
  {error&&<div className="error banner">{error}<button onClick={()=>setError('')}>Dismiss</button></div>}
  {page==='Dashboard'&&<><section className="hero"><div><span className="eyebrow light">YOUR SOCIAL LISTENING WORKSPACE</span><h1>{t('Make sense of every conversation.','हर बातचीत को समझें।')}</h1><p>Analyze audience feedback, discover recurring issues, and decide what to improve next.</p><button className="white-btn" onClick={()=>setPage('New Analysis')}>Start new analysis <ArrowUpRight size={16}/></button></div><div className="hero-art"><div className="orbit"><Activity size={76}/></div><span className="floating-chip">AI insights</span><span className="floating-chip second">Sentiment trends</span></div></section>
   <div className="stat-grid">{[['Total analyses',stats?.total_analyses??0,ScanSearch],['Comments analyzed',stats?.total_comments??0,MessageCircle],['Positive comments',stats?.positive??0,Smile],['Negative comments',stats?.negative??0,Frown]].map(([l,v,I])=><div className="stat-card" key={l}><div className="stat-top"><span>{l}</span><I size={18}/></div><strong>{v.toLocaleString()}</strong><small>Across your saved analyses</small></div>)}</div>
   <div className="section-head"><div><h3>Recent analyses</h3><p className="muted">Your latest social media reports</p></div><button className="text-btn" onClick={()=>setPage('Analysis History')}>View history <ArrowUpRight size={15}/></button></div>
   <HistoryTable items={history.slice(0,5)} open={openReport} remove={remove}/>
   <div className="notice"><b>Data access matters</b><p>Real platform analysis requires approved API access. Use Demo analysis to explore the workflow with clearly labelled synthetic sample comments.</p></div>
  </>}
  {page==='New Analysis'&&<div className="content-grid"><section className="panel"><div className="panel-title"><div><h3>Analyze a social post</h3><p className="muted">Paste a public post or video URL to begin.</p></div><span className="tag">URL ANALYSIS</span></div><form onSubmit={analyze}><label>Social media URL</label><input className="url-input" type="url" required placeholder="https://www.youtube.com/watch?v=..." value={url} onChange={e=>setUrl(e.target.value)}/><div className="platform-row"><span>Supported platforms</span><div className="platform-pills"><span>YouTube</span><span>Instagram*</span><span>Facebook*</span><span>X*</span><span>Reddit*</span></div></div><label className="check-row"><input type="checkbox" checked={demo} onChange={e=>setDemo(e.target.checked)}/> Use demo comments (synthetic sample data)</label><p className="helper">{demo?demoHelp:'Live mode fetches only content available through configured official APIs. YouTube requires YOUTUBE_API_KEY.'}</p><button className="primary" disabled={loading||!url}>{loading?<><LoaderCircle className="spin" size={17}/> Analyzing...</>:'Analyze URL'}</button></form></section><section className="panel preview-panel"><div className="panel-title"><div><h3>What you will receive</h3><p className="muted">A structured report from accessible comments.</p></div><Activity/></div><div className="preview-metric"><span className="dot green"></span><div><b>Sentiment breakdown</b><small>Positive, negative and neutral</small></div></div><div className="preview-metric"><span className="dot red"></span><div><b>Audience pain points</b><small>Recurring themes and complaints</small></div></div><div className="preview-metric"><span className="dot blue"></span><div><b>Actionable recommendations</b><small>Prioritized next steps</small></div></div><div className="preview-note">* Other platform integrations need their own developer credentials and permissions.</div></section></div>}
  {page==='Analysis'&&<>{result?<><div className="result-heading"><div><span className="eyebrow">ANALYSIS REPORT #{result.id}</span><h1>{result.title||'Social media analysis'}</h1><p className="muted">{result.platform} · {new Date(result.created_at).toLocaleString()} · {result.total_comments} comments</p></div><button className="secondary" onClick={()=>downloadPdf(result.id)}><FileDown size={17}/> Download PDF</button></div>{result.demo&&<div className="notice"><b>Demo analysis</b><p>{demoHelp}</p></div>}<div className="result-grid"><div className="panel"><h3>Original content</h3>{result.embed_url?<div className="video-wrap"><iframe src={result.embed_url} title="Original video" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture" allowFullScreen/></div>:<div className="empty-preview"><PlayCircle size={35}/><p>Preview is unavailable for this platform or sample.</p><a href={result.url} target="_blank" rel="noreferrer">Open original post <ExternalLink size={14}/></a></div>}<p className="muted small">{result.description?.slice(0,350)}</p></div><div className="panel"><h3>Sentiment distribution</h3><div className="chart-box"><ResponsiveContainer width="100%" height="100%"><PieChart><Pie data={pie} dataKey="value" nameKey="name" innerRadius={62} outerRadius={92} paddingAngle={4}>{pie.map(x=><Cell key={x.name} fill={x.color}/>)}</Pie><Tooltip/></PieChart></ResponsiveContainer><div className="chart-center"><b>{result.total_comments}</b><small>comments</small></div></div><div className="legend">{pie.map(x=><span key={x.name}><i style={{background:x.color}}/>{x.name} <b>{result.total_comments?Math.round(x.value/result.total_comments*100):0}%</b></span>)}</div></div></div><div className="panel"><h3>Comment analysis</h3><div className="comment-list">{result.comments?.slice(0,30).map((c,i)=><div className="comment-row" key={i}><span className={'sentiment '+c.sentiment}>{c.sentiment}</span><p>{c.text}</p><small>{Math.round(c.confidence*100)}% confidence</small></div>)}</div></div><div className="panel"><h3>AI recommendations</h3><div className="recommend-grid">{result.recommendations?.map((r,i)=><div className="recommend-card" key={i}><div className="recommend-top"><b>{r.problem_title}</b><span className={'priority '+r.priority}>{r.priority} priority</span></div><p>{r.problem_description}</p><b>Suggested action</b><p>{r.suggested_action}</p><small>Expected outcome: {r.expected_outcome}</small></div>)}</div></div></>:<div className="empty panel">Choose a report from Analysis History or create a new analysis.</div>}</>}
  {page==='Analysis History'&&<><div className="section-head"><div><h3>All analyses</h3><p className="muted">Open a report or remove an old record.</p></div></div><HistoryTable items={history} open={openReport} remove={remove}/></>}
  {page==='AI Recommendations'&&<><div className="panel"><h3>Decision support</h3><p className="muted">Recommendations are generated from negative comments in your analyses.</p>{result?.recommendations?.length?result.recommendations.map((r,i)=><div className="recommend-card wide" key={i}><b>{r.problem_title}</b><p>{r.problem_description}</p><p>{r.suggested_action}</p><span className={'priority '+r.priority}>{r.priority} priority</span></div>):<div className="empty">Open an analysis report to see its recommendations here.</div>}</div></>}
  {page==='AI Assistant'&&<div className="panel chat-panel"><div><h3>SocialPulse Assistant</h3><p className="muted">Ask how to use the app or understand your analysis.</p></div><div className="faq-row">{['How does sentiment analysis work?','How do I analyze a YouTube URL?','How are recommendations generated?'].map(q=><button className="faq" key={q} onClick={()=>setChatText(q)}>{q}</button>)}</div><div className="chat-messages">{chat.map((m,i)=><div className={'bubble '+m.role} key={i}>{m.text}</div>)}</div><form className="chat-form" onSubmit={askBot}><input placeholder="Ask SocialPulse Assistant..." value={chatText} onChange={e=>setChatText(e.target.value)}/><button className="primary"><Send size={17}/></button></form></div>}
  {page==='Settings'&&<div className="panel settings-panel"><h3>Preferences</h3><div className="setting-row"><div><b>Appearance</b><p className="muted">Choose your preferred interface theme.</p></div><button className="secondary" onClick={()=>setDark(!dark)}>{dark?<Sun size={16}/>:<Moon size={16}/>} {dark?'Light mode':'Dark mode'}</button></div><div className="setting-row"><div><b>Language</b><p className="muted">Select the interface language.</p></div><select value={lang} onChange={async e=>{setLang(e.target.value);try{await api('/settings',{method:'PUT',body:JSON.stringify({language:e.target.value})})}catch{}}}><option value="en">English</option><option value="hi">Hindi</option></select></div><div className="setting-row"><div><b>Guided tour</b><p className="muted">Replay the first-time application walkthrough.</p></div><button className="secondary" onClick={()=>{setStep(0);setOnboarding(true)}}>Replay tutorial</button></div><div className="setting-row"><div><b>Account</b><p className="muted">{user?.email}</p></div><button className="danger-btn" onClick={()=>{localStorage.removeItem('sp_token');setAuth(false)}}>Log out</button></div></div>}
  <footer>SocialPulse AI <span>•</span> AI-powered social media intelligence <span>•</span> Academic project edition</footer>
  </main>
  {onboarding&&<div className="modal-backdrop"><div className="tour-card"><span className="eyebrow">GETTING STARTED · STEP {step+1} OF 4</span><h2>{['Welcome to SocialPulse AI','Start with a social URL','Explore sentiment insights','Turn insights into action'][step]}</h2><p>{['Your workspace helps you understand public audience reactions and discover what to improve.','Paste a supported public post URL on the New Analysis page. Live retrieval depends on official platform API permissions.','Review positive, negative and neutral results, comments and sentiment distribution charts.','Read the recommendations, save reports and ask the assistant when you need help.'][step]}</p><div className="tour-progress"><i style={{width:`${(step+1)*25}%`}}/></div><div className="tour-actions"><button className="text-btn" onClick={finishTour}>Skip tour</button><button className="primary" onClick={()=>step===3?finishTour():setStep(step+1)}>{step===3?'Get started':'Next'}</button></div></div></div>}
 </div>
}
function HistoryTable({items,open,remove}){return <div className="table-wrap"><table><thead><tr><th>Analysis</th><th>Platform</th><th>Comments</th><th>Sentiment</th><th>Date</th><th>Actions</th></tr></thead><tbody>{items.map(a=><tr key={a.id}><td><b>{a.title||a.url?.slice(0,42)||'Analysis'}</b><small>{a.url}</small></td><td><span className="tag">{a.platform}</span></td><td>{a.total_comments}</td><td><span className="mini-pos">{a.positive_count} +</span> <span className="mini-neg">{a.negative_count} −</span> <span className="mini-neu">{a.neutral_count} =</span></td><td>{new Date(a.created_at).toLocaleDateString()}</td><td><button className="icon-btn" onClick={()=>open(a.id)} title="Open"><ArrowUpRight size={16}/></button><button className="icon-btn" onClick={()=>remove(a.id)} title="Delete"><Trash2 size={16}/></button></td></tr>)}</tbody></table>{!items.length&&<div className="empty">No analyses yet. Start by submitting a social media URL.</div>}</div>}
export default App;
